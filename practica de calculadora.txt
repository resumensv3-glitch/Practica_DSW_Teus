fun main() {
    val A: Int? = null
    val B: Int? = 5
    val operacional = "multiplicacion"

    if (A == null || B == null) {
        println("Error: Debe ingresar un valor para A y B antes de continuar")
    } else {
        if (operacional == "suma") {
            println("La suma de $A y $B es: ${A + B}")
        } 
        else if (operacional == "resta") {
            println("La resta de $A y $B es: ${A - B}")
        } 
        else if (operacional == "division") {
            if (B != 0) {
                println("La division de $A y $B es: ${A / B}")
            } else {
                println("No se puede dividir entre cero")
            }
        } 
        else if (operacional == "multiplicacion") {
            println("La multiplicacion de $A y $B es: ${A * B}")
        } 
        else {
            println("Operación no válida")
        }
    }
}
