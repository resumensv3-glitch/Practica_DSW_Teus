fun main() {
    
    val calificacion = arrayOf(9, 8, 7, 6, 10)
    val promedio = calificacion.map { it.toDouble() }.average()
    
       
    for (i in calificacion.indices)
    {
        
        println("Indice: $i, Las calificaciones son = ${calificacion[i]}")
               
    }
    
    
    
    println("Y el promedio de las calificaciones son: $promedio")
    
}
